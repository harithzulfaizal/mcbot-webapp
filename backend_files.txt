import chromadb
from datetime import datetime

def migrate_source_column():
    """
    This script adds a 'source' column to the metadata of existing documents
    in a ChromaDB collection. The source is formatted as 'FOMC Meeting Minutes {date}'.
    """
    db_path = "./chroma-data"
    collection_name = "fomc_chunk_256_metadata"

    print(f"Connecting to ChromaDB at '{db_path}'...")
    client = chromadb.PersistentClient(path=db_path)

    try:
        print(f"Getting collection '{collection_name}'...")
        collection = client.get_collection(name=collection_name)
    except Exception as e:
        print(f"Error getting collection: {e}")
        return

    print("Fetching existing documents...")
    results = collection.get(include=["metadatas"])
    
    ids = results['ids']
    metadatas = results['metadatas']
    
    if not ids:
        print("No documents found in the collection. Nothing to migrate.")
        return

    print(f"Found {len(ids)} document chunks to update.")

    new_metadatas = []
    for metadata in metadatas:
        date_val = metadata.get("date")
        if date_val:
            try:
                date_str = str(date_val)
                datetime.strptime(date_str, "%Y%m%d")
                source = f"FOMC Meeting Minutes {date_str}"
                metadata['source'] = source
            except ValueError:
                print(f"Skipping document with invalid date format: {date_val}")
        else:
            print("Skipping document with no date in metadata.")

        new_metadatas.append(metadata)

    print("Updating documents with new metadata...")
    try:
        collection.update(
            ids=ids,
            metadatas=new_metadatas
        )
        print("Migration successful. All documents have been updated with a 'source' field.")
    except Exception as e:
        print(f"An error occurred during the update: {e}")

if __name__ == "__main__":
    migrate_source_column()

@router.post("/chat/message")
async def post_message(message_input: ChatMessageInput):
    """
    Receives a chat message and enqueues it for the specific user_id and session_id.
    """
    try:
        user_id = message_input.user_id
        session_id = message_input.session_id
        session_key = (user_id, session_id)

        if not user_id or not session_id:
            raise HTTPException(status_code=400, detail="user_id and session_id are required.")

        msg_dict_to_enqueue = {
            "content": message_input.content,
            "source": message_input.source,
            "type": message_input.type,
        }
        
        await session_message_queues[session_key].put(msg_dict_to_enqueue)
        logger.info(f"Message enqueued for user {user_id}, session {session_id}: {message_input.content}")

        return {
            "status": "Message received and enqueued",
            "user_id": user_id,
            "session_id": session_id,
        }
    except Exception as e:
        logger.error(f"POST /chat/message error: {str(e)}")
        raise HTTPException(status_code=500, detail=f"Internal server error: {str(e)}")

@router.post("/kb/upload_documents")
async def upload_documents(request: Request):
    try:
        form = await request.form()
        user_id = form.get("user_id")
        files = form.getlist("files")
        if not user_id or not files:
            raise HTTPException(status_code=400, detail="user_id and files are required")

        upload_dir = f"./uploads/{user_id}"
        try:
            os.makedirs(upload_dir, exist_ok=True)
        except Exception as e:
            logger.error(f"Failed to create upload directory: {e}")
            raise HTTPException(status_code=500, detail="Failed to create upload directory")

        async def process_file(file):
            filename = file.filename
            file_path = os.path.join(upload_dir, filename)
            try:
                content = await file.read()
                with open(file_path, "wb") as f:
                    f.write(content)

                md = MarkItDown()
                result = md.convert(file_path)
                
                return Document(
                    text=result.text_content,
                    metadata={
                        "doc_id": filename,
                        "date": int(datetime.now().strftime("%Y%m%d")),
                        "source": filename,
                    },
                )
            except Exception as e:
                logger.error(f"Failed to process file {filename}: {e}")
                return None

        tasks = [process_file(file) for file in files]
        processed_documents = await asyncio.gather(*tasks)
        
        documents = [doc for doc in processed_documents if doc]
        saved_files = [doc.metadata["doc_id"] for doc in documents]

        if not documents:
            return {"user_id": user_id, "uploaded_files": [], "message": "No documents were processed successfully."}

        client = ChromaDB(db_path="./chroma-data")
        collection_name = "fomc_chunk_256_metadata"

        await asyncio.to_thread(
            client.add_documents,
            collection_name=collection_name,
            documents=documents,
            chunk_size=256,
            embed_model=embed_model,
            llm=llm,
        )

        return {"user_id": user_id, "uploaded_files": saved_files}
    except HTTPException:
        raise
    except Exception as e:
        logger.error(f"Unexpected error during upload: {e}")
        import traceback
        traceback.print_exc()
        raise HTTPException(status_code=500, detail="Unexpected error during upload")

EXTRACT_DATE_PROMPT = """
Your task is to extract and highlight the start and end dates from the query if the user specifies a specific date range. 
For a single date, use it as both the start and end.
For a year-only mention, set the start to January 1st and the end to December 31st of that year.
For a range from one year to another, start on January 1st of the first year and end on December 31st of the last year.
If no date is specified, return null for both dates.

The output should be a markdown code snippet formatted in the following schema, including the leading and trailing "```json" and "```":
```json
{{
    "start_date": string // the start date: 'yyyy-mm-dd' or null,
    "end_date": string // the end date: 'yyyy-mm-dd' or null,
}}
```

TEXT:
{query}

OUTPUT:
"""


def extract_dates(query: str) -> tuple[Optional[str], Optional[str]]:
    """
    This function extracts the start and end dates from a user's query, if they are specified.
    It takes user query as input and return the start and end dates.
    """
    # prompt to extract dates from user query
    prompt = EXTRACT_DATE_PROMPT.format(query=query)

    # get the extracted dates from LLM
    dates = LLM().call_llm(prompt)

    # verify date format
    if dates and isinstance(dates, dict):
        start_date = dates.get("start_date")
        end_date = dates.get("end_date")
    else:
        start_date, end_date = None, None

    return start_date, end_date

    """
    collection_name = "fomc_chunk_256_metadata"  # change

            messages[-1].source == planning_agent.name
            and messages[-1].type == "ToolCallSummaryMessage"

            messages[-1].source == reviewer_agent.name
            and messages[-1].type == "TextMessage"

            return summarizer_agent.name

        return None

    team = SelectorGroupChat(
        [
            planning_agent,

            reviewer_agent,

            user_proxy,

        model_client=model_client,

from llama_index.core.node_parser import TokenTextSplitter
from llama_index.embeddings.azure_openai import AzureOpenAIEmbedding
from llama_index.llms.azure_inference import AzureAICompletionsModel
from llama_index.vector_stores.chroma import ChromaVectorStore


class ChromaDB:
    """
    This class manages interactions with a Chroma-based vector database, allowing for the creation and retrieval of vector store indices.
    """

    def __init__(self, db_path: str):
        # Initializes a ChromaDB instance with a specified database path
        self.chromadb_client = chromadb.PersistentClient(path=db_path)

    def create_vector_store_index(
        self,
        collection_name: str,
        documents: list,
        chunk_size: int,
        embed_model: AzureOpenAIEmbedding,
        llm: AzureAICompletionsModel,
    ):
        """Creates a new vector store index by processing documents and adding them to a collection in the Chroma database."""
        try:
            text_splitter = TokenTextSplitter(chunk_size=chunk_size, chunk_overlap=20)

            chroma_collection = self.chromadb_client.create_collection(
                name=collection_name
            )
            vector_store = ChromaVectorStore(chroma_collection=chroma_collection)
            storage_context = StorageContext.from_defaults(vector_store=vector_store)

            qa_extractor = QuestionsAnsweredExtractor(questions=2, llm=llm)
            # keyword_extractor = KeywordExtractor(keywords=10, llm=llm)

            vector_index = VectorStoreIndex.from_documents(
                documents=documents,
                # transformations=[text_splitter, qa_extractor, keyword_extractor],
                transformations=[text_splitter, qa_extractor],
                storage_context=storage_context,
                show_progress=True,
                embed_model=embed_model,
            )

            print(
                f"Added {chroma_collection.count()} documents into collection '{collection_name}'."
            )

            return vector_index

        except Exception as e:
            print(e)

    def get_vector_store_index(
        self, collection_name: str, embed_model: AzureOpenAIEmbedding
    ) -> VectorStoreIndex:
        """Retrieves a vector store index from a specified collection within the Chroma database."""
        try:
            chroma_collection = self.chromadb_client.get_collection(
                name=collection_name
            )
            vector_store = ChromaVectorStore(chroma_collection=chroma_collection)
            vector_index = VectorStoreIndex.from_vector_store(
                vector_store,
                embed_model=embed_model,
            )

        except Exception as e:
            print(e)

        return vector_index

    def get_latest_document_date(self, collection_name: str) -> str:
        """
        Retrieves the date of the last document from a collection within the Chroma database.
        """
        try:
            chroma_collection = self.chromadb_client.get_collection(
                name=collection_name
            )
            date = str(chroma_collection.get()["metadatas"][-1]["date"])

        except Exception as e:
            print(e)

        return date

    def add_documents(
        self,
        documents: list,
        collection_name: str,
        chunk_size: int,
        embed_model: AzureOpenAIEmbedding,
        llm: AzureAICompletionsModel,
    ):
        """
        Add new document(s) to a collection within the Chroma database.
        """
        text_splitter = TokenTextSplitter(chunk_size=chunk_size, chunk_overlap=20)

        try:
            chroma_collection = self.chromadb_client.get_collection(
                name=collection_name
            )
        except Exception as e:
            print(f"Error retrieving collection {collection_name} from Chroma DB: {e}")
            return

        vector_store = ChromaVectorStore(chroma_collection=chroma_collection)
        storage_context = StorageContext.from_defaults(vector_store=vector_store)
        vector_index = VectorStoreIndex.from_vector_store(
            vector_store,
            embed_model=embed_model,
        )

        qa_extractor = QuestionsAnsweredExtractor(questions=2, llm=llm)
        # keyword_extractor = KeywordExtractor(keywords=10, llm=llm)

        vector_index._transformations = [
            text_splitter,
            qa_extractor,
        ]
        vector_index._storage_context = storage_context
        vector_index._show_progress = True

        for doc in documents:
            vector_index.insert(document=doc)

        print(f"Added {len(documents)} documents into collection '{collection_name}'.")

        return vector_index
    
    def create_collection(self, collection_name: str):
        """
        Create a new collection in the Chroma database.
        """
        try:
            self.chromadb_client.create_collection(name=collection_name)
            print(f"Collection '{collection_name}' created successfully.")
        except Exception as e:
            print(f"Error creating collection {collection_name}: {e}")

    def list_collections(self):
        """
        List all collections in the Chroma database.
        """
        try:
            collections = self.chromadb_client.list_collections()
            return collections
        except Exception as e:
            print(f"Error listing collections: {e}")
            return []
    def list_documents(self, collection_name):
        """
        List all documents in a collection within the Chroma database.
        """
        try:
            collection = self.chromadb_client.get_collection(collection_name)
            results = collection.get(include=["metadatas"])
            
            doc_info = {}
            for metadata in results["metadatas"]:
                doc_id = metadata.get("doc_id")
                if doc_id:
                    if doc_id not in doc_info:
                        doc_info[doc_id] = {
                            "date": metadata.get("date"),
                            "source": metadata.get("source"),
                            "chunk_count": 0
                        }
                    doc_info[doc_id]["chunk_count"] += 1
            
            # Convert to list of dicts
            doc_list = [{"doc_id": k, **v} for k, v in doc_info.items()]
            # Sort by doc_id
            return sorted(doc_list, key=lambda x: x['doc_id'])

        except Exception as e:
            print(f"Error listing documents: {e}")
            return []

    def delete_documents(self, collection_name, doc_id: str):
        """
        Delete existing document(s) from a collection within the Chroma database.
        """
        try:
            collection = self.chromadb_client.get_collection(collection_name)
            collection.delete(where={"doc_id": doc_id})
            print(f"Deleted document with doc_id '{doc_id}' from collection '{collection_name}'.")
        except Exception as e:
            print(f"Error deleting documents: {e}")

class LLM:
    def __init__(self):
        # Get environment variables with fallbacks
        llm_model = os.getenv("LLM_MODEL", "qwen/qwen3-8b:free")
        llm_base_url = os.getenv("LLM_BASE_URL", "")
        llm_api_key = os.getenv("LLM_API_KEY", "")
        
        if not llm_api_key:
            raise ValueError("LLM_API_KEY environment variable is not set")
            
        self.llm = OpenAILike(
            model=llm_model,
            api_base=llm_base_url,
            api_key=llm_api_key,
            context_window=32000,
            is_chat_model=True,
            is_function_calling_model=True,
            # api_version="2024-05-01-preview",
            # engine="Qwen/Qwen2.5-14B-Instruct"
        )

    def call_llm(self, content: str):
        response = self.llm.chat(
            messages=[
                ChatMessage(role="system", content="You are a helpful assistant."),
                ChatMessage(role="user", content=content),
            ],
        )
        return json_repair.loads(response.message.blocks[0].text)

    def get_llm(self) -> OpenAILike:
        return self.llm

import pandas as pd
from llama_index.core import Document


def preprocess_data() -> list:
    """
    concatenate meeting minutes, group by date & convert into Documents object
    """
    df = pd.read_csv("./data/Fed_Scrape-2015-2023.csv", encoding="utf-8", index_col=0)

    # retrieve document type "1: minutes"
    df = df[df["Type"] == 1].copy()

    # reformat date
    df["Date"] = pd.to_datetime(df["Date"], format="%Y%m%d")

    # consolidate document into one row
    df = df.groupby(["Date", "Type"])["Text"].agg("\n".join).reset_index()

    # reformat to yyyymmdd & int for metadata filtering later
    df["Date"] = df["Date"].astype(str).str.replace("-", "").astype(int)

    documents = []

    for index, row in df.iterrows():
        # create Document object
        documents.append(
            Document(
                text=row["Text"],
                metadata={
                    "doc_id": index,
                    "date": row["Date"],
                },
            )
        )

    print(f"Created {len(documents)} documents.")
    return documents

from typing import Optional

from llama_index.core import VectorStoreIndex
from llama_index.core.tools import QueryEngineTool
from llama_index.core.vector_stores.types import (
    FilterOperator,
    MetadataFilter,
    MetadataFilters,
)

from core.llm import LLM


class QueryEngine:
    def __init__(self):
        """Initialize the QueryEngine with a language model."""
        self.llm = LLM().get_llm()
        self.query_engine = None

    def create_query_engine(
        self,
        vector_index: VectorStoreIndex,
        k: int,
        start_date: Optional[str] = None,
        end_date: Optional[str] = None,
    ):
        """
        This function create a query engine using a vector store index, with an optional date filter.

        Parameters:
        - vector_index: VectorStoreIndex
            The index to query against.
        - k: int
            The number of top similar documents to retrieve.
        - start_date: Optional[str]
            The start date to filter documents.
        - end_date: Optional[str]
            The end date to filter documents.
        """
        filters = []

        if start_date:
            start_date_int = int(start_date.replace("-", ""))
            filters.append(
                MetadataFilter(
                    key="date", value=start_date_int, operator=FilterOperator.GTE
                )
            )
        if end_date:
            end_date_int = int(end_date.replace("-", ""))
            filters.append(
                MetadataFilter(
                    key="date", value=end_date_int, operator=FilterOperator.LTE
                )
            )

        # Query Data
        self.query_engine = vector_index.as_query_engine(
            similarity_top_k=k, filters=MetadataFilters(filters=filters), llm=self.llm
        )

    def create_query_engine_tool(self, name: str, description: str) -> QueryEngineTool:
        """
        Create and return a QueryEngineTool using the current query engine.

        Returns:
        - An instance of QueryEngineTool initialized with the current query engine.
        """
        if not self.query_engine:
            raise ValueError(
                "Query engine is not initialized. Please create it by calling create_query_engine()."
            )
        return QueryEngineTool.from_defaults(
            query_engine=self.query_engine,
            name=name,
            description=description,
        )

    def execute_query(self, query: str):
        """
        Execute a query using the query engine.

        Returns:
        - The result of the query as a string.
        """
        if not self.query_engine:
            raise ValueError(
                "Query engine is not initialized. Please create it by calling create_query_engine()."
            )
        try:
            results = self.query_engine.query(query)
        except Exception as e:
            print(e)
        return results

def identify_chronological_query(
    query: str,
) -> tuple[bool, list[str], Optional[str], Optional[str]]:
    """
    This function checks whether a user query involves a chronological issue and provides relevant details.

    Returns:
        - is_chronological: True if the query is a chronological issue.
        - subquery: A list of subqueries extracted from the query.
        - start_date: The start date for filtering results during document retrieval.
        - end_date: The end date for filtering results during document retrieval.

    Raises:
        - Exception: If the response is not a dictionary.
    """
    prompt = CHRONOLOGICAL_PROMPT.format(query=query)

    response = LLM().call_llm(prompt)

    # verify response format
    if response and isinstance(response, dict):
        is_chronological = bool(response.get("is_chronological"))
        subquery = response.get("subquery", [])
        start_date = response.get("start_date")
        end_date = response.get("end_date")

    else:
        raise Exception("Error dissecting user query!")

    return is_chronological, subquery, start_date, end_date


def retrieve_documents(
    source: str,
    query: str,
    collection_name: str,
    k: int,
    start_date: Optional[str] = None,
    end_date: Optional[str] = None,
) -> str:
    """
    This function filters and retrieves documents from a vector store using specified dates, and combines them into a coherent context.

    Parameters:
        - query: The query string to process.
        - collection_name: The collection name in Chroma DB.
        - k: The number of top similar results to retrieve.
        - start_date: The start date for filtering results (optional).
        - end_date: The end date for filtering results (optional).

    Returns:
        - A consolidated string of relevant documents.
    """
    # define chroma db path
    chromadb_client = ChromaDB(db_path=CHROMA_DB_PATH)

    # get the date of the last doc
    latest_doc_date = chromadb_client.get_latest_document_date(collection_name)
    print(f"Latest document date: {latest_doc_date}")

    # validate the start and end dates from PlanningAgent
    if start_date is not None and int(start_date.replace("-", "")) > int(
        latest_doc_date
    ):
        print(
            f"[Error] start_date '{start_date}' is invalid. Replacing with the latest document date '{latest_doc_date}'..."
        )
        start_date = latest_doc_date
    if end_date is not None and int(end_date.replace("-", "")) > int(latest_doc_date):
        print(
            f"[Error] end_date '{end_date}' is invalid. Replacing with the latest document date '{latest_doc_date}'..."
        )
        end_date = latest_doc_date

    vector_index = chromadb_client.get_vector_store_index(
        collection_name=collection_name, embed_model=embed_model
    )

    # query data
    query_engine = QueryEngine()
    query_engine.create_query_engine(
        vector_index=vector_index,
        k=k,
        start_date=start_date,
        end_date=end_date,
    )
    retrieved_docs = query_engine.execute_query(query)

    context = f"""
    # Context ({source})
    """

    for i in range(k):
        context += f"""
    Source: {source}, Date: {retrieved_docs.source_nodes[i].node.metadata["date"]}, Content: {retrieved_docs.source_nodes[i].node.text}

    """

    return context


def retrieve_fomc_documents(
    query: str, start_date: Optional[str], end_date: Optional[str]
) -> str:
    """
    This function filters and retrieves documents from the FOMC vector store using specified dates, and combines them into a coherent context.

    Parameters:
        - query: The query string to process.
        - start_date: The start date for filtering results (optional).
        - end_date: The end date for filtering results (optional).

    Returns:
        - A consolidated string of relevant documents from FOMC vector database.
    """

    return retrieve_documents(
        source=SOURCE_NAME_FOMC,
        collection_name=COLLECTION_NAME_FOMC,
        query=query,
        k=K,
        start_date=start_date,
        end_date=end_date,
    )


def generate_response(query: str, context: str):
    """
    This function generates a response using the context from retrieved documents.

    Parameters:
        - query: The query string to process.
        - context: A consolidated string of relevant documents.

    Returns:
        - The response generated by LLM.
    """
    prompt = GENERATE_ANSWER_PROMPT.format(context=context, query=query)

    # get the response from LLM
    response = LLM().call_llm(prompt)
    return response

                <h1 class="welcome-title">MCBot Chat Assistant</h1>
                <p class="welcome-subtitle">How can I help you today?</p>
                <div class="welcome-suggestions">
                    <div class="suggestion-item" onclick="useSuggestion(this)">Tell me about federal rate decision</div>
                    <div class="suggestion-item" onclick="useSuggestion(this)">How did the decision to maintain the rate come to a conclusion?</div>
                </div>
            </div>
            <div id="messages"></div>
        </div>
        
        <div class="chat-footer">
            <div class="input-container-wrapper">
                <div id="input-container">
                    <textarea id="message-input" placeholder="Type Your Message..." rows="1" onkeydown="handleKeyDown(event)" oninput="this.style.height = ''; this.style.height = Math.min(this.scrollHeight, 150) + 'px'"></textarea>
                    <div class="input-buttons">
                        <button class="input-action-btn" title="Attach files">
                            <i class="fas fa-paperclip"></i>
                        </button>
                        <button id="send-button" onclick="sendMessage()" title="Send message">
                            <i class="fas fa-paper-plane"></i>
                        </button>
                    </div>
                </div>
                <div class="input-footer">
                    MCBot may produce inaccurate information about people, places, or facts.
                </div>
            </div>
        </div>
    </div>

    <script>
        // WebSocket connection
        // const ws = new WebSocket('ws://localhost:8000/api/ws/chat');

        // SSE
        const eventSource = new EventSource('http://localhost:8000/api/chat/events');

        // Initialize page state
        let welcomeScreen = document.getElementById('welcome-screen');
        let messagesContainer = document.getElementById('messages');
        let messageInput = document.getElementById('message-input');
        let sendButton = document.getElementById('send-button');
        
        // Basic functions
        function handleKeyDown(event) {
            if (event.key === 'Enter' && !event.shiftKey && !messageInput.disabled) {
                event.preventDefault();
                sendMessage();
            }
        }
        
        function useSuggestion(element) {
            messageInput.value = element.textContent;
            messageInput.style.height = '';
            messageInput.style.height = Math.min(messageInput.scrollHeight, 150) + 'px';
            sendMessage();
        }
        
        function sendMessage() {
            const message = messageInput.value.trim();
            if (!message) return;
            
            // Clear input and disable
            messageInput.value = '';
            messageInput.style.height = '';
            disableInput();
            
            // Hide welcome screen
            if (welcomeScreen.style.display !== 'none') {
                welcomeScreen.style.display = 'none';
            }
            
            // Display user message
            // displayMessage(message, 'user');
            
            // Add typing indicator
            addTypingIndicator();
            
            // Send to WebSocket
            // ws.send(JSON.stringify({ content: message, source: 'user' }));

            // To send a chat message (using fetch or any library):
            fetch('http://localhost:8000/api/chat/message', {
                method: 'POST',
                headers: {
                'Content-Type': 'application/json',
                },
                body: JSON.stringify({
                content: message,
                source: 'user'
                }),
            });
        }
        
        function addTypingIndicator() {
            const typingDiv = document.createElement('div');
            typingDiv.className = 'typing-indicator';
            typingDiv.innerHTML = `
                <div class="label">
                    <div class="label-avatar">
                        <i class="fas fa-robot"></i>
                    </div>
                    <span>Assistant is typing</span>
                </div>
                <div class="typing-dots">
                    <div class="typing-dot"></div>
                    <div class="typing-dot"></div>
                    <div class="typing-dot"></div>
                </div>
            `;
            messagesContainer.appendChild(typingDiv);
            scrollToBottom();
        }
        
        async function clearHistory() {
            try {
                const response = await fetch('http://localhost:8000/api/history/clear');
                if (!response.ok) {
                    throw new Error('Failed to clear history');
                }
                // Clear messages
                messagesContainer.innerHTML = '';
                // Show welcome screen
                welcomeScreen.style.display = 'flex';
                displayMessage("Chat history cleared.", 'system');
            } catch (error) {
                console.error('Error clearing history:', error);
                displayMessage("Failed to clear history: " + error.message, 'error');
            }
        }
        
        function displayMessage(content, source) {
            const messageElement = document.createElement('div');
            messageElement.className = `message ${source}`;
            
            // Create label
            const labelElement = document.createElement('div');
            labelElement.className = 'label';
            
            let labelIcon = '';
            let labelText = '';
            
            switch(source) {
                case 'user':
                    labelIcon = '<div class="label-avatar">U</div>';
                    labelText = 'You';
                    break;
                case 'assistant':
                    labelIcon = '<div class="label-avatar"><i class="fas fa-robot"></i></div>';
                    labelText = 'Assistant';
                    break;
                case 'system':
                    labelIcon = '<div class="label-avatar"><i class="fas fa-info-circle"></i></div>';
                    labelText = 'System';
                    break;
                case 'error':
                    labelIcon = '<div class="label-avatar"><i class="fas fa-exclamation-triangle"></i></div>';
                    labelText = 'Error';
                    break;
                default:
                    labelIcon = '<div class="label-avatar"><i class="fas fa-comment"></i></div>';
                    labelText = source;
            }
            
            labelElement.innerHTML = `${labelIcon}<span>${labelText}</span>`;
            
            // Create content
            const contentElement = document.createElement('div');
            contentElement.className = 'content';
            
            // Check if content is JSON
            let isJson = false;
            let jsonContent;
            try {
                if (typeof content === 'object' && content !== null) {
                    jsonContent = content;
                    isJson = true;
                } else if (typeof content === 'string') {
                    jsonContent = JSON.parse(content);
                    isJson = true;
                }
            } catch (e) {
                isJson = false;
            }
            
            if (isJson) {
                const jsonContainer = document.createElement('pre');
                jsonContainer.className = 'json-content';
                const formattedJson = JSON.stringify(jsonContent, null, 2);
                jsonContainer.textContent = formattedJson;
                contentElement.appendChild(jsonContainer);
            } else {
                contentElement.innerHTML = window.marked ? marked.parse(content) : content;
            }
            
            // Create action buttons
            let actionsDiv = null;
            if (source === 'user' || source === 'assistant') {
                actionsDiv = document.createElement('div');
                actionsDiv.className = 'message-actions';
                
                if (source === 'user') {
                    actionsDiv.innerHTML = `
                        <button class="message-action-btn" onclick="editMessage(this)">
                            <i class="fas fa-edit"></i> Edit
                        </button>
                    `;
                } else if (source === 'assistant') {
                    actionsDiv.innerHTML = `
                        <button class="message-action-btn" onclick="copyMessage(this, '${content.replace(/'/g, "\\'")}')">
                            <i class="fas fa-copy"></i> Copy
                        </button>
                        <button class="message-action-btn">
                            <i class="fas fa-thumbs-up"></i>
                        </button>
                        <button class="message-action-btn">
                            <i class="fas fa-thumbs-down"></i>
                        </button>
                    `;
                }
            }
            
            // Assemble message
            messageElement.appendChild(labelElement);
            messageElement.appendChild(contentElement);
            if (actionsDiv) {
                messageElement.appendChild(actionsDiv);
            }
            
            // Add to container and scroll
            messagesContainer.appendChild(messageElement);
            scrollToBottom();
        }
        
        function editMessage(button) {
            const messageElement = button.closest('.message');
            const contentElement = messageElement.querySelector('.content');
            const messageText = contentElement.textContent.trim();
            
            // Set the text to input and focus
            messageInput.value = messageText;
            messageInput.focus();
            messageInput.dispatchEvent(new Event('input'));
        }
        
        function copyMessage(button, text) {
            navigator.clipboard.writeText(text).then(() => {
                // Show copied feedback
                const originalHTML = button.innerHTML;
                button.innerHTML = '<i class="fas fa-check"></i> Copied';
                setTimeout(() => {
                    button.innerHTML = originalHTML;
                }, 2000);
            });
        }
        
        function disableInput() {
            messageInput.disabled = true;
            sendButton.disabled = true;
            sendButton.innerHTML = '<span class="loading-spinner"></span>';
        }
        
        function enableInput() {
            messageInput.disabled = false;
            sendButton.disabled = false;
            sendButton.innerHTML = '<i class="fas fa-paper-plane"></i>';
            messageInput.focus();
        }
        
        function scrollToBottom() {
            messagesContainer.scrollTop = messagesContainer.scrollHeight;
        }
        
        // event handlers
        eventSource.onmessage = function(event) {
            const message = JSON.parse(event.data);
            
            // Remove typing indicator
            const typingIndicator = document.querySelector('.typing-indicator');
            if (typingIndicator) {
                typingIndicator.remove();
            }
            
            // if (message.type === 'UserInputRequestedEvent') {
            //     enableInput();
            // }
            // else if (message.type === 'ModelResponse') {
            if (message.type === 'ModelResponse') {
                // Create collapsible container
                const collapsible = document.createElement('details');
                const summary = document.createElement('summary');
                
                summary.textContent = message.source;
                
                // Create content container
                const contentContainer = document.createElement('div');
                contentContainer.className = 'content';
                contentContainer.innerHTML = window.marked ? marked.parse(message.content) : message.content;
                
                // Add classes for styling
                collapsible.classList.add('message');
                collapsible.classList.add('system-message');
                
                // Assemble the elements
                collapsible.appendChild(summary);
                collapsible.appendChild(contentContainer);
                
                // Add to messages container
                messagesContainer.appendChild(collapsible);
                scrollToBottom();

                enableInput();
            }
            else if (message.type === 'error') {
                displayMessage(message.content, 'error');
                enableInput();
            }
            else {
                displayMessage(message.content, message.source);
            }
        };
        
        eventSource.onerror = function(error) {
            displayMessage("SSE error occurred. Please refresh the page.", 'error');
            enableInput();
        };
        
        eventSource.onclose = function() {
            displayMessage("Connection closed. Please refresh the page.", 'system');
            disableInput();
        };
        
        // Handle sidebar toggle
        const menuToggle = document.querySelector('.menu-toggle');
        const sidebar = document.querySelector('.sidebar');
        
        menuToggle.addEventListener('click', function() {
            sidebar.classList.toggle('open');
        });
        
        // Handle dark mode toggle
        const toggleThemeButton = document.getElementById('toggle-theme');
        toggleThemeButton.addEventListener('click', function() {
            document.body.classList.toggle('dark-mode');
            const icon = toggleThemeButton.querySelector('i');
            if (document.body.classList.contains('dark-mode')) {
                icon.className = 'fas fa-sun';
            } else {
                icon.className = 'fas fa-moon';
            }
        });
        
        // Handle chat menu items
        const chatItems = document.querySelectorAll('.chat-item');
        chatItems.forEach(item => {
            item.addEventListener('click', function() {
                // Skip if this is the theme toggle
                if (this.id === 'toggle-theme') return;
                
                // Remove active class from all items
                chatItems.forEach(i => {
                    if (i.id !== 'toggle-theme') {
                        i.classList.remove('active');
                    }
                });
                
                // Add active class to clicked item
                this.classList.add('active');
                
                // Update header title
                document.querySelector('.chat-header-title span').textContent = this.querySelector('.chat-title').textContent;
                
                // Close sidebar on mobile
                if (window.innerWidth <= 768) {
                    sidebar.classList.remove('open');
                }
            });
        });
        
        // New chat button
        document.querySelector('.new-chat-btn').addEventListener('click', function() {
            clearHistory();
            
            // Set active class to the new session item
            chatItems.forEach(i => {
                if (i.id !== 'toggle-theme') {
                    i.classList.remove('active');
                }
            });
            
            // Find New Session item and make it active
            const newSessionItem = document.querySelector('.chat-item:nth-child(2)');
            if (newSessionItem) {
                newSessionItem.classList.add('active');
                document.querySelector('.chat-header-title span').textContent = newSessionItem.querySelector('.chat-title').textContent;
            }
        });
        
        // Load chat history on page load
        async function loadHistory() {
            try {
                const response = await fetch('http://localhost:8000/api/history');
                if (!response.ok) {
                    throw new Error('Network response was not ok');
                }
                const history = await response.json();
                
                // If there's history, hide the welcome screen
                if (history.length > 0) {
                    welcomeScreen.style.display = 'none';
                }
                
                history.forEach(message => {
                    displayMessage(message.content, message.source);
                });
                
                // If no history, make sure welcome screen is visible
                if (history.length === 0) {
                    welcomeScreen.style.display = 'flex';
                }
            } catch (error) {
                console.error('Error loading history:', error);
                displayMessage("Failed to load chat history. Starting a new session.", 'system');
            }
        }
        
        // Load history when the page is ready
        loadHistory();
    </script>
